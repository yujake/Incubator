===== G-10：CancelReason.code 的受控词表 =====
CANCEL_CODES = ('user', 'deadline', 'budget', 'dispose', 'cancelled')
CANCEL_BY_VALUES = ('unknown', 'user', 'deadline', 'budget', 'conductor.dispose')
-- 正控 1：词表内的 code 照常可用
    CancelReason('user') ok -> {'code': 'user', 'detail': 'd', 'outcome_known': False, 'by': 'unknown'}
    CancelReason('deadline') ok -> {'code': 'deadline', 'detail': 'd', 'outcome_known': False, 'by': 'unknown'}
    CancelReason('budget') ok -> {'code': 'budget', 'detail': 'd', 'outcome_known': False, 'by': 'unknown'}
    CancelReason('dispose') ok -> {'code': 'dispose', 'detail': 'd', 'outcome_known': False, 'by': 'unknown'}
    CancelReason('cancelled') ok -> {'code': 'cancelled', 'detail': 'd', 'outcome_known': False, 'by': 'unknown'}
-- 正控 2：Conductor._signal_stop 那一层早就拒词表外的值（本来就成立）
    conductor.py 含 `if code not in CANCEL_CODES` = True
-- 负控 1：直接构造词表外的 code 必须抛（修前通过）
    rejected CancelReason('随便写的') -> CancelReason.code='随便写的' 不在受控词表 ('user', 'deadline', 'budget
    rejected CancelReason('reason-7') -> CancelReason.code='reason-7' 不在受控词表 ('user', 'deadline', 'bu
    rejected CancelReason('other') -> CancelReason.code='other' 不在受控词表 ('user', 'deadline', 'budge
-- 负控 2：CancelToken.cancel(词表外的 code) 必须抛（修前通过）
    rejected cancel('reason-7') -> CancelReason.code='reason-7' 不在受控词表 ('user', 'deadline', 'bu
    失败后 token 状态：cancelled = False ，reason = None （必须仍是未取消 —— 不许留下半取消状态）
-- 边界：`by` 刻意仍不是硬枚举（调用方自定义标签必须被接受）
    by = caller-with-my-own-name ；在词表里 = False
-- 边界：幂等/first-wins 不受影响（第二次的 code 不再被构造）
    第二次返回 == 第一次 = True ；code = budget
-- 端到端：detail_of 渲染
    取消[budget] 预算耗尽
